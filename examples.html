Examples

Copy-paste snippets for common tasks. All assume import MetalTerrain.

Minimal app (UIKit)

import MetalTerrain

class ViewController: UIViewController {
    var terrainView: MTTerrainView!

    override func viewDidLoad() {
        super.viewDidLoad()
        terrainView = MTTerrainView(frame: view.bounds)
        terrainView.autoresizingMask = [.flexibleWidth, .flexibleHeight]
        terrainView.world = MTTerrainWorld(seed: 1337)
        view.addSubview(terrainView)

        // Safe spawn for the player
        let spawn = terrainView.world!.findSafeSpawn()
        let groundY = terrainView.world!.worldY(
            forHeight: terrainView.world!.heightAt(x: Double(spawn.x), z: Double(spawn.y)))

        terrainView.onFrame = { [weak self] tv, dt in
            guard let self else { return }
            let aspect = Float(tv.bounds.width / tv.bounds.height)
            tv.setCamera(position: [spawn.x, groundY + 60, spawn.y + 120],
                         target: [spawn.x, groundY, spawn.y],
                         aspect: aspect)
        }
    }
}

Orbit camera (drag to rotate, pinch to zoom)

var yaw: Float = -0.6, pitch: Float = 0.62, distance: Float = 550
var target = SIMD3<Float>(0, 0, 0)

terrainView.onFrame = { tv, dt in
    let aspect = Float(tv.bounds.width / tv.bounds.height)
    let cp = cos(pitch)
    let pos = target + SIMD3<Float>(sin(yaw) * cp, sin(pitch), cos(yaw) * cp) * distance
    tv.setCamera(position: pos, target: target, aspect: aspect)
}
// Wire yaw/pitch/distance to your gesture recognizers:
//   drag → yaw -= dx * 0.0055; pitch = clamp(pitch - dy * 0.0055, 0.08, 1.35)
//   pinch → distance = clamp(distance / scale, 40, 1200)

Walk mode (first-person, water blocks movement)

var playerPos = world.findSafeSpawn()
var walkYaw: Float = 0, walkPitch: Float = -0.1
let eyeHeight: Float = 2.0

terrainView.onFrame = { tv, dt in
    // moveInput: SIMD2<Float> from your joystick (x = strafe, y = forward)
    let speed: Float = 45
    let fwd = SIMD2<Float>(sin(walkYaw), cos(walkYaw))
    let right = SIMD2<Float>(-fwd.y, fwd.x)
    let delta = (fwd * moveInput.y + right * moveInput.x) * speed * Float(min(dt, 0.1))

    // Don't walk into the ocean
    let tryPos = playerPos + delta
    if world.heightAt(x: Double(tryPos.x), z: Double(tryPos.y)) >= world.config.seaLevel {
        playerPos = tryPos
    }

    let groundY = world.worldY(forHeight: world.heightAt(x: Double(playerPos.x), z: Double(playerPos.y)))
    let eye = SIMD3<Float>(playerPos.x, groundY + eyeHeight, playerPos.y)
    let cp = cos(walkPitch)
    let look = SIMD3<Float>(sin(walkYaw) * cp, sin(walkPitch), cos(walkYaw) * cp)
    let aspect = Float(tv.bounds.width / tv.bounds.height)
    tv.setCamera(position: eye, target: eye + look * 10, aspect: aspect)
}

Full app skeleton (UIKit)

The canonical 60-line integration — view, world, camera, sun slider:

import UIKit
import MetalTerrain

final class TerrainViewController: UIViewController {
    var terrainView: MTTerrainView!
    var world: MTTerrainWorld!
    // Camera state
    var yaw: Float = 0.8, pitch: Float = -0.45, distance: Float = 1400
    var target = SIMD3<Float>(0, 120, 0)

    override func viewDidLoad() {
        super.viewDidLoad()
        world = MTTerrainWorld(seed: 20261005)
        terrainView = MTTerrainView(frame: view.bounds)
        terrainView.autoresizingMask = [.flexibleWidth, .flexibleHeight]
        terrainView.world = world
        // Spawn looking at land, not ocean:
        let s = world.findSafeSpawn()
        target = SIMD3<Float>(s.x, world.worldY(
            forHeight: world.heightAt(x: Double(s.x), z: Double(s.y))), s.y)
        terrainView.onFrame = { [weak self] view, dt in
            self?.updateCamera(view: view, dt: dt)
        }
        view.addSubview(terrainView)
    }

    func updateCamera(view: MTTerrainView, dt: TimeInterval) {
        let eye = SIMD3<Float>(
            target.x + distance * cos(pitch) * sin(yaw),
            target.y + distance * sin(pitch),
            target.z + distance * cos(pitch) * cos(yaw))
        let aspect = Float(view.bounds.width / max(1, view.bounds.height))
        view.setCamera(position: eye, target: target, aspect: aspect)
    }
}

Walk-mode player controller

// Joystick input in [-1, 1]²; call from onFrame.
func updatePlayer(view: MTTerrainView, world: MTTerrainWorld,
                  pos: inout SIMD2<Float>, yaw: inout Float,
                  moveInput: SIMD2<Float>, dt: Float) {
    let speed: Float = 45  // world units/sec at full tilt
    let forward = SIMD2<Float>(sin(yaw), cos(yaw))
    let right = SIMD2<Float>(-forward.y, forward.x)
    var delta = (forward * moveInput.y + right * moveInput.x) * speed * dt

    // Water blocking: don't walk into the ocean (slide along the shore).
    func landAt(_ p: SIMD2<Float>) -> Bool {
        world.heightAt(x: Double(p.x), z: Double(p.y)) >= world.config.seaLevel
    }
    let tryPos = pos + delta
    if landAt(tryPos) { pos = tryPos }
    else if landAt(SIMD2(pos.x + delta.x, pos.y)) { pos.x += delta.x }
    else if landAt(SIMD2(pos.x, pos.y + delta.y)) { pos.y += delta.y }

    // Eye follows terrain; sample ahead so hills don't swallow the camera.
    let ground = world.worldY(forHeight: world.heightAt(x: Double(pos.x),
                                                        z: Double(pos.y)))
    let ahead = pos + forward * 8
    let aheadY = world.worldY(forHeight: world.heightAt(x: Double(ahead.x),
                                                         z: Double(ahead.y)))
    let eye = SIMD3<Float>(pos.x, max(ground, aheadY) + 2, pos.y)
    let look = eye + SIMD3<Float>(forward.x, -0.08, forward.y) * 10
    let aspect = Float(view.bounds.width / max(1, view.bounds.height))
    view.setCamera(position: eye, target: look, aspect: aspect)
}

Regenerate button (debounced)

// Coalesce rapid taps: stale builds are dropped via buildGeneration,
// but don't queue hundreds of them in the first place.
var lastRegen = Date.distantPast
@IBAction func regenerateTapped() {
    guard Date().timeIntervalSince(lastRegen) > 1.5 else { return }
    lastRegen = Date()
    world.seed = UInt64.random(in: 1...999_999)
    // configVersion is unchanged by seed... force a stream refresh:
    world.config = world.config  // bumps configVersion → renderer rebuilds
    let s = world.findSafeSpawn()
    target = SIMD3<Float>(s.x, 0, s.y)
}

Quality settings screen

func applyQuality(_ q: Quality, view: MTTerrainView) {
    switch q {
    case .low:
        view.viewDistance = 4
        view.renderer?.detailAmount = 0.3
        view.showsWater = true
    case .medium:
        view.viewDistance = 6
        view.renderer?.detailAmount = 0.7
    case .high:
        view.viewDistance = 8
        view.renderer?.detailAmount = 1.0
    }
    // viewDistance writes through to world.config → streaming rebuilds.
}
// Changing viewDistance at runtime is safe: the renderer evicts out-of-range
// chunks and streams the new ring nearest-first.

Procedural minimap

Because generation is deterministic, a minimap is just sampling:

func renderMinimap(world: MTTerrainWorld, center: MTChunkCoord,
                   radius: Int, pixelsPerChunk: Int) -> UIImage {
    let size = (radius * 2 + 1) * pixelsPerChunk
    let renderer = UIGraphicsImageRenderer(size: CGSize(width: size, height: size))
    return renderer.image { ctx in
        for cz in -radius...radius {
            for cx in -radius...radius {
                let chunk = world.generateChunk(
                    at: MTChunkCoord(x: center.x + cx, z: center.z + cz),
                    resolutionScale: 0.25)  // cheap preview resolution
                for iz in 0..<chunk.resolution {
                    for ix in 0..<chunk.resolution {
                        let h = chunk.height(ix: ix, iz: iz)
                        let b = world.biomeAt(height: h)
                        let c = b.groundColor
                        ctx.cgContext.setFillColor(
                            red: CGFloat(c.x), green: CGFloat(c.y),
                            blue: CGFloat(c.z), alpha: 1)
                        let px = (cx + radius) * pixelsPerChunk
                               + ix * pixelsPerChunk / chunk.resolution
                        let py = (cz + radius) * pixelsPerChunk
                               + iz * pixelsPerChunk / chunk.resolution
                        ctx.cgContext.fill(CGRect(x: px, y: py, width: 2, height: 2))
                    }
                }
            }
        }
    }
}
